       ```python
       def handle_save_selection_set(params):
           """Save the current selection under a name for later recall."""
           obj = _get_mesh_object(params["object_name"])
           set_name = params["set_name"]
           if obj.mode != "EDIT":
               raise ValueError(
                   f"Object '{obj.name}' has no active edit-mode selection to save. "
                   "Select something first (select_vertices_by_index / select_edges_by_index / select_faces_by_index)."
               )
           sets = _load_selection_sets(obj)
           sets[set_name] = _current_selection_by_type(obj)
           _save_selection_sets(obj, sets)
           return {"object": obj.name, "set_name": set_name, "saved": sets[set_name]}

       def handle_select_selection_set(params):
           """Recall a previously-saved named selection set."""
           obj = _get_mesh_object(params["object_name"])
           set_name = params["set_name"]
           sets = _load_selection_sets(obj)
           if set_name not in sets:
               raise ValueError(
                   f"Selection set '{set_name}' not found on '{obj.name}'. "
                   f"Available: {sorted(sets.keys())}"
               )
           saved = sets[set_name]

           _select_only(obj)
           bpy.ops.object.mode_set(mode="EDIT")
           bpy.ops.mesh.select_all(action="DESELECT")

           bm = bmesh.from_edit_mesh(obj.data)
           for elem_type, seq_name in (("vert", "verts"), ("edge", "edges"), ("face", "faces")):
               indices = saved.get(elem_type, [])
               if not indices:
                   continue
               seq = getattr(bm, seq_name)
               seq.ensure_lookup_table()
               max_index = len(seq) - 1
               for i in indices:
                   if 0 <= i <= max_index:
                       seq[i].select = True

           bm.select_flush(True)
           bmesh.update_edit_mesh(obj.data)
           return {"object": obj.name, "set_name": set_name, "applied": saved, "mode": "EDIT"}

       def handle_list_selection_sets(params):
           """List saved selection set names on an object."""
           obj = _get_mesh_object(params["object_name"])
           sets = _load_selection_sets(obj)
           return {"object": obj.name, "set_names": sorted(sets.keys())}
       ```

       ```python
       @mcp.tool()
       def save_selection_set(object_name: str, set_name: str) -> dict[str, Any]:
           """Save the current edit-mode selection under a name for later recall.

           Args:
               object_name: Name of the mesh object (must already have an active
                   edit-mode selection from select_vertices_by_index / select_edges_by_index
                   / select_faces_by_index / select_selection_set).
               set_name: Name to save the selection under.

           Returns:
               Dict confirming the saved set contents.
           """
           object_name = validate_object_name(object_name)
           set_name = validate_object_name(set_name)
           conn = get_connection()
           response = conn.send_command("save_selection_set", {
               "object_name": object_name,
               "set_name": set_name,

           if response.get("status") == "error":
               raise RuntimeError(f"Blender error: {response.get('result')}")
       ```
